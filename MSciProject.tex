\section{Introduction}
\paragraph{Context and Motivation}
Mean field games is a branch of game theory devoted to the analysis of a special subset of 'games' within the discipline. 
These systems or 'games' are distinguished by having a very large number of players, each of whom is considered almost
insignificant with minimal influence on the system as a whole. In measure theory this corresponds to nullifying sets with countable measure and 
thinking of functions in the distributional sense. Furthemore, each player is a rational observer and the general trend around them will inform their decision on how to act, 
with the aim of acheiving the most optimal outcome.
The famous example in economics of such a game, introduced by Aumann, is the behaviour of stock traders whom we treat as a continuum \cite{aumann}.
\\
In this paper we consider the equilibrium problem characterised by a coupled pair of equations. These are a Hamilton--Jacobi--Bellman (HJB) equation with a Kolmogorov--Fokker--Planck (KFP) equation

\begin{equation}
\label{eq:system}
\begin{aligned}
-\Delta u + H(x,\nabla u) + \lambda u &= f(x,m) && \text{in } \Omega \text{ (HJB)},\\
-\Delta m - \diver\!\bigl(m\,\Hp(x,\nabla u)\bigr) + \lambda m &= G(x) && \text{in } \Omega \text{ (KFP)},
\end{aligned}
\end{equation}
together with homogeneous Dirichlet boundary conditions $u=0$ and $m=0$ on $\partial\Omega$.

\paragraph{Interpretation of Equations}
$m(x)$ represents the player density and $u(x)$ the value function of a representative agent in the mean field game. In other words, $u(x)$ is a player's optimal solution achieved by minimising the expected cost at position $x$.
Here $\Hp$ denotes the partial derivative of $H$ with respect to the momentum variable (i.e.\ the $\nabla u$ argument), and $\lambda>0$.
\\
(HJB) describes a representative player's optimal control behaviour for a given player density, $m$, in the game. Whereas the (KFP) equation takes the macroscopic view and gives us the evolution of $m(\cdot)$ when every individual player satisfies the optimal control $u$.\\
Each player tries to minimise their cost function $f(x,m)$ (in HJB) and the optimal control for an individual player is given by $\alpha^* = -\Hp (x, \nabla u)$ which we can see makes up the (KFP).
The $\Delta$ terms are called 'diffusion’ terms which account for the randomness in the game (determined by Brownian motion) and prevent the perfect alignment of players even if they all adopt the same strategy. 
Lastly, $G()$ is an exogenous function, controlling the entry /exit of players to and from the system and as such will directly affect $m$ which is why it appears in the (KFP).
\cite{lasry_lions, pierre}

\paragraph{Mathematical Contribution}

In this paper we analyse this coupled system of equations and in particular we prove the existence of a  weak solution pair $(u,m)$ in $\Honez \times \Honez$, provided the following conditions are met:
\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=0.35em]
  \item $\Omega$ is bounded and $\partial\Omega$ is smooth.
  \item $H(x,\nabla u):\R^{n+1}\to\R$ is
  \begin{itemize}[leftmargin=*, itemsep=0.25em]
    \item Lipschitz continuous with respect to $\nabla u$ with Lipschitz constant $\tilde{C}$,
    \item bounded by $C\bigl(|x|+|\nabla u|+1\bigr)$ for a given constant $C$, for all $x\in\R$ and $\nabla u\in\R^n$.
  \end{itemize}
  \item The partial derivative $H_p$ is continuous with respect to $\nabla u$ and essentially bounded by some constant $L$.
  \item $\lambda$ is sufficiently large (in particular $\lambda>\max(\,\frac{L^2}{4},\,\frac{C^2}{2},\,\frac{\tilde{C}^2}{4})$).
  \item $f(x,m(x)):(\Omega,\Ltwo)\to \Ltwo$ and is $\Ltwo$-continuous.
  \item $G(x)\in \Ltwo$.
\end{enumerate}

\section{Method}

Our proof consists of two stages. First, we demonstrate solvability for each equation separately, while fixing the other variable (HJB with fixed $m$, and KFP with fixed $u$).
Second, we show the existence of a solution \emph{pair} solving both equations simultaneously.

\subsection{Independent solutions}
Before launching forward it's important to introduce the following inequalities. These are essential and will lend themselves
to almost every proof throughout this paper.

\begin{definition}
  $p,\,q \in [1, \infty)$ are \emph{conjugate parameters} if ${\frac{1}{q}+ \frac{1}{p} = 1}$ 
\end{definition}
\begin{theorem}[Hölder's inequality]
\label{thm:holder}
Let $f \in L^p(\Omega) \text{ and } g \in L^q(\Omega)$ where $p, \,q$ are conjugate parameters. Then
\[\norm{fg}_1 \le \norm{f}_p\,\norm{g}_q\]
\end{theorem}
\begin{remark}
  In particular, when $p,\,q = 2$ we get the Cauchy--Schwarz inequality
  \[\norm{fg}_1 \le \norm{f}_2\,\norm{g}_2\]
\end{remark}
\begin{theorem}[Young's inequality]
\label{thm:youngs}
Let $a,\, b \in \R$ then 
\[2ab \le \frac{a^2}{\varepsilon^2} + \varepsilon^2 b^2 \qquad \forall \varepsilon > 0\]
\end{theorem}
\subsubsection{Solving Kolmogorov--Fokker--Planck}

To solve
\[
-\Delta m - \diver\!\bigl(m\,\Hp(x,\nabla u)\bigr) + \lambda m = G(x)
\]
with fixed $u$, we note that the equation is linear in $m(x)$, so we appeal to the Lax--Milgram theorem \cite{lax_milgram}.

\begin{theorem}[Lax--Milgram]
\label{thm:lax-milgram}
Let $V$ be a Hilbert space and let $a:V\times V\to\R$ be a bilinear form such that for all $u,v\in V$,
\begin{enumerate}[label=(\roman*), leftmargin=*, itemsep=0.25em]
  \item (boundedness) $|a(u,v)|\le c_1\norm{u}\,\norm{v}$,
  \item (coercivity) $a(v,v)\ge c_0\norm{v}^2$.
\end{enumerate}
Let $l:V\to\R$ be a bounded linear functional. Then there exists a unique $u\in V$ such that
\[
a(u,v)=l(v)\qquad\text{for all }v\in V.
\]
Moreover, $\norm{u}_V \le \frac{1}{c_0}\norm{l}_{V^\ast}$.
\end{theorem}

For our purposes, we take $a(m,v)=l(v)$ for all $v\in\Honez$ to be the weak formulation of (KFP), i.e.
\[
a(m,v)=\int_\Omega \nabla m\cdot\nabla v + m\,\nabla v\cdot\Hp(x,\nabla u)+\lambda mv\,dx,
\qquad
l(v)=\int_\Omega G(x)\,v\,dx.
\]

\begin{proposition}
\label{prop:kfp-bounded}
The form $a(m,v)$ is bounded on $\Honez\times\Honez$.
\end{proposition}
\begin{proof} 
For arbitrary $m,v\in\Honez$,
\begin{align*}
|a(m,v)|
&\le \int_{\Omega} |\nabla m||\nabla v| +|\nabla v||m||\Hp| + \lambda |m||v| \,dx\\
&\le \norm{\nabla m}_{L^2}\norm{\nabla v}_{L^2} + L\norm{m}_{L^2}\norm{\nabla v}_{L^2} + \lambda\norm{m}_{L^2}\norm{v}_{L^2}\\
&\le C\,\norm{m}_{H^1}\,\norm{v}_{H^1}.
\end{align*}
Making use of the Cauchy--Schwarz inequality in the second line.
\end{proof}

\begin{proposition}
\label{prop:kfp-coercive}
If $\lambda$ is sufficiently large, then (KFP) is coercive.
\end{proposition}
\begin{proof}
Consider
\begin{align*}
a(v,v)
&\ge \norm{\nabla v}_{L^2}^2 - L\norm{v}_{L^2}\norm{\nabla v}_{L^2} + \lambda\norm{v}_{L^2}^2\\
&\ge \Bigl(1-\frac{\varepsilon^2}{2}\Bigr)\norm{\nabla v}_{L^2}^2
  + \Bigl(\lambda-\frac{L^2}{2\varepsilon^2}\Bigr)\norm{v}_{L^2}^2,
  \qquad \forall \varepsilon\in\R,
\end{align*}
by Young's inequality. To ensure $1-\varepsilon^2/2>0$, impose $\varepsilon^2<2$.
Choosing $\varepsilon^2=2$ (as a limiting choice) gives the weakest requirement
\[\lambda>\frac{L^2}{4},\]
which ensures coercivity.
\end{proof}

\begin{remark}
Both proofs use Condition 3, namely $\lvert H_p\rvert\le L$.
\end{remark}

Thus $a(\cdot,\cdot)$ is bounded and coercive. Next,
\begin{align*}
\norm{l}
&= \sup_{\norm{v}_{H^1}\le 1}\left|\int_\Omega Gv\,dx\right|
\le \sup_{\norm{v}_{L^2}\le 1}\norm{v}_{L^2}\,\norm{G}_{L^2}
= \norm{G}_{\Ltwo},
\end{align*}
so $l$ is bounded. By Theorem \ref{thm:lax-milgram}, there exists a unique solution $m\in\Honez$.

\begin{remark}
This method applies here because (KFP) is linear in $m$ once $u\in\Honez$ is fixed.
\end{remark}

\begin{definition}
\label{def:S-map}
Define $S:\Honez\to\Honez$ by $S(u)=m$, where $m$ solves (KFP) for fixed $u$.
\end{definition}

\subsubsection{Solving Hamilton--Jacobi--Bellman}

Lax--Milgram does not apply directly to
\[-\Delta u + H(x,\nabla u) + \lambda u = f(x,m)\]
with fixed $m$, so we use fixed point methods. Throughout this section, $X$ denotes a real Banach space.

\begin{theorem}[Schauder's fixed point theorem]
If $K\subset X$ is compact and convex, and $A:K\to K$ is continuous, then $A$ has a fixed point in $K$.
\end{theorem}

\begin{definition}
A mapping $A:X\to X$ is called \emph{compact} if for every bounded sequence $(u_k)_{k=1}^\infty$,
the sequence $(A[u_k])_{k=1}^\infty$ is precompact in $X$.
\end{definition}

\begin{theorem}[Schaefer's fixed point theorem]
\label{thm:schaefer}
Suppose $A:X\to X$ is continuous and compact, and assume the set
\[
\{u\in X: u=\mu A[u]\ \text{for some }0\le\mu\le 1\}
\]
is bounded. Then $A$ has a fixed point.
\end{theorem}
\footnote{See \cite[Chapter~9]{evans_pde} for proofs.}

\begin{theorem}[Existence]
\label{thm:existence}  
If $\lambda>0$ is sufficiently large, there exists $u\in H^2(\Omega)\cap \Honez$ solving (HJB) with fixed $m$.
\end{theorem}
\begin{proof}
Consider
\[
A:\Honez\to\Honez,\qquad A[u]=w
\quad\text{where}\quad
-\Delta w+\lambda w = f(x,m)-H(x,\nabla u).
\]
We know $w$ exists by Lax--Milgram. We apply Theorem \ref{thm:schaefer}.
Continuity and compactness are as in \cite[Chapter~9]{evans_pde} (with the addition of an $\norm{f}_{\Ltwo}$ term), so we omit them.
It remains to show
\[\{u\in \Honez: u=\mu A[u]\ \text{for some }0\le\mu\le 1\}\]
is bounded.

Substituting $w=u/\mu$, we obtain
\begin{align*}
\int_\Omega |\nabla u|^2 + \lambda |u|^2\,dx
&= \mu\int_\Omega f u - H(x,\nabla u)u\,dx\\
&\le \int_\Omega |fu| + |H(x,\nabla u)u|\,dx\\
&\le \frac12\int_\Omega |fu| + C(|x|+|\nabla u|+1)|u|\,dx \qquad \text{(Condition 2)}\\
&\le \frac12\int_\Omega \frac{f^2}{\varepsilon^2} + \varepsilon^2 u^2
 + \frac{1}{\delta^2}|Cu|^2 + \delta^2(|x|+|\nabla u|+1)^2\,dx,\\
&\hspace{7.3cm}\varepsilon\in (0, \infty),\ \delta\in(0,1)\\
&\le \frac12\int_\Omega \frac{f^2}{\varepsilon^2}
 + \Bigl(\varepsilon^2+\frac{C^2}{\delta^2}\Bigr)u^2
 + \delta^2(1+\gamma^2)|\nabla u|^2\,dx + K(\gamma), \qquad \gamma\in (0, \infty),
\end{align*}
after repeated applications of Young's inequality to isolate $|\nabla u|^2$ and $u^2$.
We require $\delta^2(1+\gamma^2)<1$ (in particular sufficiently small $\delta^2<1$ suffices for any $\gamma >0$).
Since $\varepsilon$ can be arbitrarily small, this yields a bound on $\norm{u}_{\Honez}$ provided $\lambda>\frac{C^2}{2}$.
\end{proof}
\begin{remark}
  We can always tune $\varepsilon, \delta \text{ and } \gamma$ for given $\lambda>\frac{C^2}{2}$ to satisfy boundedness. This utilises the
  strictness of the inequality.
\end{remark}
\begin{proposition}
\label{prop:uniqueness}
If $\lambda>0$ is sufficiently large, then the fixed point $u$ is unique.
\end{proposition}
\begin{proof}
Suppose $u_1,u_2$ both solve (HJB). Taking the difference gives
\begin{align*}
\norm{\nabla(u_1-u_2)}_{L^2}^2+\lambda\norm{u_1-u_2}_{L^2}^2
&\le \int_\Omega |u_1-u_2|\,|H_1-H_2|\,dx\\
&\le \tilde{C}\,\norm{u_1-u_2}_{L^2}\,\norm{\nabla(u_1-u_2)}_{L^2}\\
&\le \frac12 \bigl(\frac{\tilde{C}^{\,2}}{\varepsilon^2}\,\norm{u_1-u_2}^2_{L^2} +\varepsilon^2 \norm{\nabla(u_1-u_2)}^2_{L^2}\bigr)
\qquad \varepsilon\in (0, \sqrt{2})\\
\end{align*}
using Condition 2 and Cauchy--Schwarz. Hence, by Young's inequality,
\[\lambda>\frac{\tilde{C}^{\,2}}{4}\implies u_1=u_2 \text{ in }\Ltwo\iff u_1=u_2 \text{ in }\Honez.\]
\end{proof}

\begin{definition}
\label{def:T-map}
Define $T:\Honez\to\Honez$ by $T(m)=u$, where $u$ solves (HJB) for fixed $m$.
\end{definition}

\subsection{Codependent solution}

An equivalent formulation of finding a solution pair $(u,m)$ is to find a fixed point of a suitable map.

\begin{definition}
Let $T$ and $S$ be as in definitions \ref{def:S-map} and \ref{def:T-map}. Define
\[A:\Honez\to\Honez,\qquad A \coloneqq S\circ T.\]
\end{definition}

That is, $A$ takes $m\in\Honez$, produces $u=T(m)$ solving (HJB), and then produces $\widetilde m=S(u)$ solving (KFP).
If $A$ has a fixed point, then $m=\widetilde m = A(m)$, and the coupled system has a solution pair.

We verify the hypotheses of Schaefer's theorem.

\subsubsection{Continuity of $A$}
Since $A=S\circ T$, it suffices to show $S$ and $T$ are continuous.

\paragraph{Continuity of $T$.}
\label{pf:t-cont}
Let $(m_k)_{k=1}^\infty\to m$ in $\Honez$. Denote $u=T[m]$, $u_k=T[m_k]$ and $f_k = f(x, m_k)$.
Taking the weak formulation of the difference in (HJB) gives
\begin{align*}
\norm{\nabla(u-u_k)}_{L^2}^2+\lambda\norm{u-u_k}_{L^2}^2
&\le \int_\Omega |u-u_k|\,|H-H_k| + |u-u_k|\,|f-f_k|\,dx\\
&\le \tilde{C}\norm{u-u_k}_{L^2}\norm{\nabla(u-u_k)}_{L^2}
 + \norm{u-u_k}_{L^2}\norm{f-f_k}_{L^2}.
\end{align*}
Since $f$ is $\Ltwo$-continuous (Condition 5), $\norm{f-f_k}_{L^2}\to 0$.
Also, from the energy estimate
\begin{align*}
\norm{\nabla u_k}_{L^2}^2+\lambda\norm{u_k}_{L^2}^2
&\le \int_\Omega  |u_k|\,|f_k| + |u_k|\,|H_k|\,dx.\\
\end{align*}
Reapplying the boundedness argument from \ref{thm:existence} and noting that by it's convegence to $f, \, f_k$ is bounded, we get that $(u_k)$ is bounded in $\Honez$.
Thus $\norm{u-u_k}_{L^2}\norm{f-f_k}_{L^2}\to 0$, and the uniqueness argument above (\ref{prop:uniqueness}) implies $u_k\to u$.

\paragraph{Continuity of $S$.}
\label{pf:s-cont}
Let $(u_k)_{k=1}^\infty\to u$ in $\Honez$. Denote $m=S[u]$, $m_k=S[u_k]$ and $\Hp_k=\Hp(x,\nabla u_k)$.
Taking the weak formulation of the difference in (KFP) yields
\begin{align*}
\norm{\nabla(m-m_k)}_{L^2}^2+\lambda\norm{m-m_k}_{L^2}^2
&\le \int_\Omega |\nabla(m-m_k)|\,\bigl|m\,\Hp - m_k\,\Hp_k\bigr|\,dx.
\end{align*}

Define the correction factor $\varepsilon_k \coloneqq \Hp - \Hp_k$. Then, 
\begin{align*}
\norm{\nabla(m-m_k)}_{L^2}^2+\lambda\norm{m-m_k}_{L^2}^2
&\le \int_\Omega |\nabla(m-m_k)|\,\bigl|m\,\Hp - m_k\, \Hp\bigr| +|m_k||\varepsilon_k|\,dx\\
&\le L \int_\Omega |\nabla(m-m_k)|\,|m - m_k|\,dx + \int_\Omega |m_k|\,|\varepsilon_k|\,dx.
\end{align*}
$m$ and $m_k$ are bounded (\ref{thm:lax-milgram}) so as $k$ gets large $|m_k||\varepsilon_k| \to 0$. Therefore, by the largeness of
$\lambda$ (see \ref{prop:kfp-coercive}), we get $m_k \to m$ i.e $S$ is continuous.

\begin{remark}
  The image of $S$ is bounded as a direct consequence of Lax-Milgram. 
\end{remark}

\subsubsection{Compactness of $A$}

Since $S$ is continuous and bounded, it suffices to show that $T$ is compact in $\Honez$:
\[T \text{ compact } \implies A=S\circ T \text{ compact.}\]

\begin{definition}
Let $X$ and $Y$ be Banach spaces with $X\subset Y$. We say that $X$ is compactly embedded in $Y$, written $X\subset\subset Y$, provided:
\begin{enumerate}[label=(\roman*), leftmargin=*, itemsep=0.25em]
  \item $\norm{x}_Y \le C\norm{x}_X$ for some constant $C$,
  \item each bounded sequence in $X$ is precompact in $Y$.
\end{enumerate}
\end{definition}

\begin{theorem}[Rellich--Kondrachov compactness theorem]
Assume $U$ is a bounded open subset of $\R^n$ and $\partial U$ is $C^1$. Suppose $1<p<n$. Then
\[W^{1,p}(U)\subset\subset L^q(U) \quad\text{for each }1\le q<p^\ast, \qquad p^\ast=\frac{np}{n-p}.\]
\end{theorem}

\begin{remark}
Observe $p^\ast>p$. In particular,
\[W^{1,p}(U)\subset\subset L^p(U) \quad\text{for all }1\le p<n.\]
\end{remark}

Let $(m_k)_{k=1}^\infty$ be a bounded sequence in $\Honez$.
Since $\Omega$ satisfies Condition 1, there exists a subsequence $(m_{k_j})_{j=1}^\infty$ converging in $\Ltwo$ to some $m\in\Ltwo$.
We want to show $T(m_{k_j})\to T(m)$.
Because $f$ is $\Ltwo$-continuous (Condition 5), $\Ltwo$ convergence is sufficient, and the continuity argument for $T$ applies with $m_{k_j}$.
Thus $T$ is compact, and hence $A$ is compact.

\subsubsection{Boundedness of the pseudo-fixed point set}

Consider
\[\{m\in\Honez: m=\mu A[m]\ \text{for some }\mu\in[0,1]\}.\]
The corresponding system, with $A[m]$ substituted for $m/\mu$, is
\[-\Delta u + H(x,\nabla u) + \lambda u = f(x,m), \qquad \frac{1}{\mu}\Bigl(-\Delta m - \diver\!\bigl(m\,\Hp(x,\nabla u)\bigr) + \lambda m\Bigr)=G(x).\]
Rearranging the latter and using the Lax--Milgram bound implies
\[\norm{m}_{L^2}\le \mu \norm{G}_{L^2}\le \norm{G}_{L^2},\]
which is bounded in $\Ltwo$ (Condition 6). By Schaefer's theorem, $A$ has a fixed point, hence a solution pair $(u,m)$ exists.

\begin{remark}
In hindsight, we see the order of composition was convenient because it provides a natural bound for the pseudo-fixed point set.
\end{remark}

Therefore $A$ has a fixed point and a solution pair $(u,m)\in \Honez$ exists.

\subsection{Swapping Composition Order}
An equaly valid construction for $A$ is the reverse composition 
\[A \coloneqq S\circ T.\] 
$S$ and $T$ are unchanged so the compactness of $A$ still holds. 
In the proof of theorem \ref{thm:existence} we demonstrated that $\lambda > \frac{C^2}{2} \implies \norm{u}_{\Honez}$ is bounded when $m$ is fixed. 
Under this new construction we must consider 
\[\{u\in\Honez: u=\mu A[u]\ \text{for some }\mu\in[0,1]\}.\]
The corresponding system, with $A[u]$ substituted for $u/\mu$, is
\[\frac{1}{\mu}\Bigl(-\Delta u + H(x,\nabla u) + \lambda u)\Bigr) = f(x,m), \qquad -\Delta m - \diver \bigl(m\,\Hp(x,\nabla u)\bigr) + \lambda m=G(x).\]

\emph{wrong -} We note that allowing $m(x)$ to be variable makes no difference in this respect since $\norm{f(x,m(x))}_{L^2}$ is always bounded on $\Omega$ (Condition 5).
Thus the set of pseudo-fixed points is still bounded and Schaefer's Theorem carries over. 

\section{Non-linear (KFP)}
The following is an analysis on a variation of \ref{eq:system} with the $G(\cdot)$ in question being dependent on $m$. 
Our new system of equations reads
\begin{equation}
\label{eq:system2}
\begin{aligned}
-\Delta u + H(x,\nabla u) + \lambda u &= f(x,m) && \text{in } \Omega,\\
-\Delta m - \diver\!\bigl(m\,\Hp(x,\nabla u)\bigr) + \lambda m &= G(x,m) && \text{in } \Omega.
\end{aligned}
\end{equation}

In this scenario we cannot apply the Lax-Milgram Theorem to (KFP). We will therefore adopt fixed point theorems 
to find a solution but as with $f(x,m(x))$ we must impose conditions on $G(x,m(x))$.
\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=0.35em, resume]
\item $G(x,m):(\Omega,\Ltwo)\to \Ltwo$ is
  \begin{itemize}[leftmargin=*, itemsep=0.25em]
    \item Lipschitz continuous with respect to $m$ with Lipschitz constant $\tilde{B}$,
    \item bounded by $B\bigl(|x|+|m|+1\bigr)$ for a given constant $B$, for all $x\in \Omega$ and $m \in \Ltwo$.
  \end{itemize}
\item In addition to our previous constraint on $\lambda$, we have $\lambda > \max(\,B + \frac{L^2}{4},\,\tilde{B} +\frac{L^2}{4})$
\end{enumerate}

\begin{theorem}[Existence of Non-linear KFP Solution]
\label{thm:existence2}  
If $\lambda>0$ is sufficiently large, there exists $m\in H^2(\Omega)\cap \Honez$ solving (KFP) with fixed $u$.
\end{theorem}
\begin{proof}
  See \ref{thm:existence} and \cite[Chapter~9]{evans_pde}.
  The only difference is in the thershold of $\lambda$ which we'll now derive. As in \ref{thm:existence} we want to show
  \[\{m\in \Honez: m=\mu A[m]\ \text{for some }0\le\mu\le 1\}\]
  is bounded.
  \footnote{$A(\cdot)$ here is the map defined for fixed $u$ not to be confused with the composition of $S$ and $T$.}
  Substituting $w=m/\mu$, we obtain
  \begin{align*}
  \int_\Omega |\nabla m|^2 + \lambda |m|^2\,dx
  &= \mu\int_\Omega G(x,m)m - m \nabla m \Hp  \,dx\\
  &\le \int_\Omega |G(x,m)||m| + L|m||\nabla m|\,dx\\
  &\le \int_\Omega B(|x|+|m|+1)|m| + L|m||\nabla m|\,dx \qquad \text{(Condition 7)}\\
  &\le \int_\Omega B|m|^2 + \frac12(\frac{L^2}{\varepsilon^2}|m|^2 + \varepsilon^2|\nabla m|^2) + K \,dx 
  \qquad \varepsilon\in (0, \sqrt{2})\\
  &\le \int_\Omega (B +\frac{L^2}{2 \varepsilon^2})|m|^2 + \frac12(\varepsilon^2|\nabla m|^2) + K \,dx.\\
  \end{align*}
  Once more using Young's inequality. This yields a bound on $\norm{m}_{\Honez}$ provided ${\lambda> B +\frac{L^2}{4}}$.
\end{proof}
\begin{proposition}
If $\lambda>0$ is sufficiently large, then the fixed point $m$ is unique.
\end{proposition}
\begin{proof}
Suppose $m_1,m_2$ both solve (HJB). Taking the difference gives
\begin{align*}
\norm{\nabla(m_1-m_2)}_{L^2}^2 +\lambda\norm{m_1-m_2}_{L^2}^2
&\le \int_\Omega |\Hp|\,|m_1-m_2|\,|\nabla(m_1 -m_2)| + |G_1-G_2|\,|m_1-m_2|\,dx\\
&\le L \norm{m_1-m_2}_{L^2}\norm{\nabla(m_1-m_2)}_{L^2} + \tilde{B}\,\norm{m_1-m_2}_{L^2}^2\\
\end{align*}
using Conditions 2, 7 and Cauchy--Schwarz. Hence, by our previous use of Young's inequality in \ref{prop:kfp-coercive},
\[\lambda - \tilde{B} >\frac{L^{\,2}}{4}\implies m_1=m_2 \text{ in }\Honez.\]
\end{proof}

Thus our definition of $S(\,\cdot\,)$ in \ref{def:S-map} remains well defined for (KFP) in its new form.
We have left to prove that $S$ is still continuous after which our previous results will hold for compactness of $A$.

\paragraph{Continuity of S \#2}
\label{pf:s-cont2}
Oncemore, let $(u_k)_{k=1}^\infty\to u$ in $\Honez$. Denote $m=S[u]$, $m_k=S[u_k]$, $\Hp_k=\Hp(x,\nabla u_k)$ and $G_k = G(x,m_k)$.
Taking the weak formulation of the difference in (KFP) yields
\begin{align*}
\norm{\nabla(m-m_k)}_{L^2}^2+\lambda\norm{m-m_k}_{L^2}^2
&\le \int_\Omega |\nabla(m-m_k)|\,\bigl|m\,\Hp - m_k\,\Hp_k\bigr| + |G - G_k|\,|m-m_k|\,dx\\
&\le \int_\Omega |\nabla(m-m_k)|\,\bigl|m\,\Hp - m_k\,\Hp_k\bigr| + \tilde{B}|m-m_k|^2\,dx.
\end{align*}
By $\lambda$ being sufficiently large to account for $\tilde{B}$ and following the same line of reasoning in \ref{pf:s-cont} we conclude that 
$S$ is continuous.
