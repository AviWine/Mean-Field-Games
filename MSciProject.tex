\section{Introduction}
\begin{flushleft}
\textit{(Italics are copied) Mean field games (MFG), which were introduced by Lasry and Lions [24, 25, 26] and independently by Huang, Caines, and Malham´e [22], 
model Nash equilibria of rational games of stochastic optimal control in which there are infinitely many players. The equilibria are characterized by a system of partial differential equations (PDE) that consist of
a Hamilton–Jacobi–Bellman (HJB) equation coupled with a Kolmogorov–Fokker–Planck
(KFP) equation. Comprehensive reviews of MFG, including analysis and applications, are
well-documented in the literature [3, 19, 20]. We consider as a model problem a quasilinear
elliptic stationary MFG system of the form}
$${-\Delta u + H(x, \nabla u) +\lambda u = f(x,m) \text{ in } \Omega \text{,}}$$ 
$${-\Delta m - \operatorname{div}(m \frac{\partial H}{\partial p}(x, \nabla u)) + \lambda m = G(x) \text{ in } \Omega \text{,}}$$

along with homogeneous Dirichlet boundary conditon $u=0$ and $m=0$ on $\partial \Omega$. 
These equations can be interpreted as ... with $m(x)$ representing player density and $u(x)$ the \textit{value function of the underlying optimal control problem faced by the players}. 
Here ${\frac{\partial H}{\partial p}}$ is shorthand for the partial deriative w.r.t ${\nabla u}$ and $\lambda > 0$.

In this paper I demonstrate the existence of a solution pair $(u,m)$ in  $H^1_0(\Omega)$, provided the following conditons are met: 
\begin{enumerate}
    \item $\Omega$ is bounded and $\partial \Omega$ is smooth
    \item $H(x,\nabla u) : \mathbb{R}^{n+1} \to \mathbb{R}$ is  
    \begin{itemize}
        \item Lipschitz continuous with respect to $\nabla u$ with Lipschitz constant $\tilde{C}$,  
        \item bounded by $C(|x| + |\nabla u| + 1)$ for a given constant $C$, $\forall x \in \mathbb{R}$, $\nabla u \in \mathbb{R}^n$.
    \end{itemize}
    \item The partial derivative $H_p$ is continuous with respect to $\nabla u$ and essentially bounded by some constant $L$.
    \item $\lambda$ is sufficiently large.
    \item $f(x, m(x)) : (\Omega, L^2(\Omega)) \to L^2(\Omega)$ and is $L^2$-continuous.
    \item $G(x) \in L^2(\Omega)$.
\end{enumerate}

\section{Method}
Our solution will consists of two stages. 
First we'll demonstrate solvability for each equation - fixing the second free variable - 
seprately for (HJB) and (KFP). 
Next comes the existence of a solution \emph{pair} that solves both equations simultaneously. 

\subsection{Independant Solutions}
\subsubsection{Solving Kolmogorov–Fokker–Planck}
To solve ${-\Delta m -m \operatorname{div}(\frac{\partial H}{\partial p}(x, \nabla u)) + \lambda m = G(x)}$ with fixed $u$, we note that the equation 
is linear in $m(x)$ so we call upon the famous Lax-Milgram Theorem \cite{lax_milgram}. 
\begin{theorem}[Lax--Milgram]
    Let $V$ be a Hilbert space, and let $${a : V \times V \to \mathbb{R}}$$ 
    be a bilinear form s.t for all ${u, v \in V}$, $a$ is: 
    \begin{enumerate}
        \item bounded: $|a(u,v)| \le c_1\|u\| \|v\|$
        \item coercive: $a(v,v) \ge c_0 \|v\| \|v\|$.

    \end{enumerate} Let ${l : V \to \mathbb{R}}$ be a bounded linear functional. Then there exists a unique ${u \in V}$
    such that 
    $${a(u, v) = l(v) \text{ for all } v \in V}$$. 

    Furthermore the solution $u$ satisfies ${\|u\|_{V} \le \frac{1}{c_0}\|l(v)\|}$. 
\end{theorem}

For our purposes we'll take ${a(m,v) = l(v) \text{, } \forall v \in H^1_0(\Omega)}$ to be the weak formulation of (KFP)
i.e $${a(m,v) = \int_{\Omega} \nabla m \nabla v + m \nabla v \frac{\partial H}{\partial p} + \lambda mv \, dx }$$
and $${l(v) = \int_{\Omega} G(x)v \, dx}$$

\begin{proposition}
    $a(m,v)$ is bounded.
\end{proposition}
\begin{proof}
    For arbitrary $m,v \in H^1_0(\Omega)$ we have 
    \begin{align*}
    |a(m,v)|
    &\le \|\nabla m\|_{L^2}\|\nabla v\|_{L^2} + L \|m\|_{L^2}\| \nabla v\|_{L^2} + \lambda \|m\|_{L^2}\|v\|_{L^2}\\[1em]
    &\le C \|m\|_{H^1}\|v\|_{H^1}\\
    \end{align*}
\end{proof}
\begin{proposition}
    If $\lambda$ is sufficiently large then (KFP) is coercive.
\end{proposition}
\begin{proof}
    Consider 
    \begin{align*}
    a(v,v)
    &\ge \|\nabla v\|_{L^2}^2 - L \|v\|_{L^2}\| \nabla v\|_{L^2} + \lambda \|v\|_{L^2}^2\\[1em]
    &\ge (1-\frac{\epsilon^2}{2})\|\nabla v\|_{L^2}^2 +  (\lambda-\frac{L^2}{2\epsilon^2}) \|v\|_{L^2}^2 \qquad \forall \epsilon \in \mathbb{R}
    && \text{(Young's Inequality)}\\
    \end{align*}     

    To ensure ${1-\frac{\epsilon^2}{2} > 0}$ we restrict ${\epsilon^2 <2}$ and conclude with the weakest requirement on
    $\lambda$ being 
    $${\lambda > \frac{L^2}{8}}$$ 
    which guarantees coercivity. 
\end{proof}
\begin{remark}
    Both proofs build on condition $3$ i.e that $|H_p| \le L$.
\end{remark}

$a(m,v)$ is thus bounded and coercive. 

Now consider
    \begin{align*}
    \|l(v)\|
    & = \sup_{\|v\|_{H^1} \le 1} |\int_{\Omega} Gv \, dx|\\[1em]
    & \le \sup_{\|v\|_{L^2} \le 1} \|v\|_{L^2}\|G\|_{L^2}
    && \text{(Cauchy-Schwarz)}\\[1em]
    & = \|G\|_{L^2(\Omega)}
    && \text{which is bounded (conditon 6.)}\\
    \end{align*}

so $l(v)$ is thus bounded. 
By Lax-Milgram we may conclude that a unique solution $m \in H^1_0(\Omega)$, exists.

\begin{remark}
    This method will \emph{only} work for (KFP) which becomes a linear PDE with respect to $m$ once ${u \in H^1_0(\Omega)}$ is fixed. 
\end{remark}

\begin{definition} 
    $${S: H^1_0(\Omega) \to H^1_0(\Omega)}$$  
    $${S(u) = m \text{ whereby } m \text{ solves (KFP) for fixed } u}$$ 
\end{definition}

$S$ is a well defined map since m both exists and is unique. We will make use of this map
in the next section.  

\subsubsection{Solving Hamilton–Jacobi–Bellman}

Lax-Milgram will unfortunately not do to solve ${-\Delta u + H(x, \nabla u) +\lambda u = f(x,m)}$
for fixed $m$ so we resolve to adopt the following fixed point theorems.\\
Throughout this section $X$ denotes a real Banach space.

\begin{theorem}[Schauder's Fixed Point Theorem]
    Suppose $K \subset X$ is compact and convex, and assume also 
    $$A:K \to K $$
    is continuous. Then $A$ has a fixed point in $K$.
\end{theorem}

We follow this up with it's consequence.

\begin{definition}
    A nonlinear mapping is $A: X \to X$ is called \emph{compact} provided for each bounded sequence $(u_k)_{k=1}^\infty$
    the sequence $(A[u_k])_{k=1}^\infty$ is precompact; that is, there exists a subsequence $(u_{k_j})_{k_j=1}^\infty$ such that
    $(A[u_{k_j}])_{k_j=1}^\infty$ converges in $X$.
\end{definition}

\begin{theorem}[Schaefer's Fixed Point Theorem]
Suppose
$$A:X \to X$$
is a continuous and compact mapping. Assume further that the set
$$\{u \in X |u=\mu A[u] \text{ for some } 0 \le \mu \le 1 \}$$
is bounded. Then A has a fixed point.

\end{theorem}
{\footnotesize See \cite[Chapter~9]{evans_pde} for proofs.}
    
The somewhat less restrictive nature of the fixed point theorems means they can be applied in
the semilinear PDE context.\\
Additionally, with  Schaefer's Theorem we get an added advantage over Schauder's of not having to identify a compact set. 
In our case however, we'll find that the natural bound of our map informs us on an appropriate set to choose. 

\begin{theorem}[Existence]
    If $\lambda > 0$ is sufficiently large, there exists a
    function $u \in H^2(\Omega)\cap H^1_0(\Omega)$ solving (HJB) with fixed $m$.
\end{theorem}
\begin{proof}
    We consider a map 
    $$A: H^1_0(\Omega) \to H^1_0(\Omega)$$
    $$A[u] = w \text{, such that } -\Delta w +\lambda w = f(x,m) - H(x, \nabla u)$$
    We know $w$ exists by Lax-Milgram. Our aim then is to demonstrate the existence of a fixed point of $A$ using Schaefer's Theorem.  
    Proving continuity and compactness of $A$ is almost identical to \cite[Chapter~9]{evans_pde} (except for an $\|f\|_{L^2(\Omega)}$ term) so we'll omit it.
    Left to show is that 
    $$\{u \in X |u=\mu A[u] \text{ for some } 0 \le \mu \le 1 \}$$
    is bounded.
    
    Subtituting $\frac{u}{\mu}$ for $w$, we find ourselves considering
    \begin{align*}
        \int_\Omega (\nabla u)^2 +\lambda u^2 \,dx 
        & = \mu \int_\Omega fu - H(x, \nabla u)u\,dx\\[1em]
        & \le \int_\Omega |fu| +|H(x, \nabla u)u| \,dx\\[1em]
        & \le \frac{1}{2}\int_\Omega |fu| + C(|x| + |\nabla u| + 1)|u| \, dx \qquad \text{(Condition 2.)}\\[1em]
        & \le \frac{1}{2}\int_\Omega \frac{f^2}{\epsilon^2} + \epsilon^2 u^2 + \frac{1}{\delta^2}|Cu|^2 + \delta^2(|x|+|\nabla u| + 1)^2 \,dx, \qquad \epsilon \in \mathbb{R},\text{ } \delta \in (0,1)\\[1em]
        & \le \frac{1}{2}\int_\Omega \frac{f^2}{\epsilon^2} + (\epsilon^2 + \frac{1}{\delta^2}C^2) u^2 +\delta^2 (1+\gamma^2)(\nabla u)^2 \,dx + K, \qquad \gamma \in \mathbb{R}   
    \end{align*}

    having applied Young's Inequality repeatidly to fully isolate $(\nabla u)^2$ and $u^2$. We need to ensure 
    \begin{align*}
        \delta^2(1+\gamma^2) < 1
        & \implies \delta^2 < 1
    \end{align*}
    Also, since $\epsilon$ can be arbitrarily close to $0$, this leaves $\|u\|_{H^1}$ bounded if ${\lambda > \frac{C^2}{2}}$ 
\end{proof}

\begin{proposition}
    If $\lambda > 0$ is sufficiently large then soution $u$ is unique
\end{proposition}
\begin{proof}
    Suppose $\exists u_1, u_2$ both solving the HJB equation. Then
    taking their difference yields 
    \begin{align*}
        \|\nabla(u_1 - u_2)\|_{L^2}^2 +\lambda\|(u_1-u_2)\|_{L^2}^2
        & \le \int_\Omega |(u_1-u_2)||H_1 - H_2|\,dx\\[1em]
        & \le \tilde{C}^2\|(u_1-u_2)\|_{L^2} \|\nabla(u_1-u_2)\|_{L^2}\\[1em]
        & \text{(Condition 2.) and (Cauchy-Schwarz)}
    \end{align*}
    So 
    \begin{align*}
        \lambda > \frac{\tilde{C}^4}{4}
        & \implies u_1 = u_2 \text{ in } L^2(\Omega)\\[1em]
        &&\text{(Young's Inequality)}\\[1em]
        & \iff u_1 = u_2 \text{ in } H^1_0(\Omega)\\
    \end{align*}
    Note that $\tilde{C}$ denotes the Lipschitz constant of $H(x, \nabla u)$.
\end{proof}
Once again we have both existence and uniqueness so the following is well defined.

\begin{definition}
    $${T: H^1_0(\Omega) \to H^1_0(\Omega)}$$  
    $${T(m) = u \text{ whereby } u \text{ solves (HJB) for fixed } m}$$     
\end{definition}

We will make use of $T$ in the next section

\subsection{Codependant Solution}
An equivalent formulation to the problem of finding a solution pair (u,m) is to find a fixed point of $A$. 

\begin{definition}
    Let $T$ and $S$ be the maps defined above. 
    $${A: H^1_0(\Omega) \to H^1_0(\Omega)}$$
    $${A := S \circ T}$$
\end{definition}

That is, $A$ takes an ${m \in H^1_0(\Omega)}$, internally converts it to $u \in H^1_0(\Omega)$ which solves (HJB) 
and converts that to another (possibly distinct) ${\widetilde{m}  \in H^1_0(\Omega)}$ which solves (KFP).\\[1em]
If A has a fixed point then there exists an ${m = \widetilde{m} = A(m)}$.
It is natural to look once more to the Schauder and Schaefer theorems and see whether $A$ satisifes their
conditions. 

We'll frame the conditions for A in the context of Schaefer's Theorem. 
\subsubsection{Continuity of A}
$A$ is continuous if $T$ and $S$ are both continous, so we will demonstrate the continuity of $S$ and $T$.

First consider $T$.\\[1em]
Let $(m_k)_{k=1}^\infty$ be a convergent sequence in $H^1_0(\Omega)$ and $m = \lim_{k \to \infty} m_k$.
Lable $u = T[m]$ and $u_k = T[m_k]$.\\
Taking the weak formulation of their difference in (HJB) gives
\begin{align*}
        \|\nabla(u - u_k)\|_{L^2}^2 +\lambda\|(u-u_k)\|_{L^2}^2
        & \le \int_\Omega |(u-u_k)||H - H_k| + |(u-u_k)||f - f_k| \,dx\\[1em]
        & \le \tilde{C}^2\|(u-u_k)\|_{L^2} \|\nabla(u-u_k)\|_{L^2} + \|u-u_k\|_{L^2}\|f-f_k\|_{L^2}\\[1em]
        & \text{(Condition 2.) and (Cauchy-Schwarz)}
    \end{align*}

Recall that $f$ is $L^2$ continuos (Condition 5.) so ${\|f-f_k\|_{L^2} \to 0 \text{ as } k \to \infty}$.
Note also that
\begin{align*}
    \|\nabla u_k\|_{L^2}^2 + \lambda\|u_k\|_{L^2}^2
    & \le \int_\Omega |u_k||H_k| + |u_k||f_k| \,dx\\[1em]
\end{align*}

$H$ and $f_k$ are both bounded. Applying Young's Inequality Thus $u_k$ are bounded and in particular $\|u_k - u\|_{L^2}$ is bounded.
We may say therefore 
$$\|u-u_k\|_{L^2}\|f-f_k\|_{L^2} \to 0$$ 
and repeating the argument above
$\implies u_k \to u \text{ as } k \to \infty$.\\
